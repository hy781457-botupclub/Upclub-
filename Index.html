async function fetchLive() { try { // सीधे आपके Render API से डेटा मांगना const response = await fetch('/api/live-result'); const data = await response.json(); // चेक करें कि डेटा में पीरियड और प्रेडिक्शन (Size) है या नहीं if (data.next_period && data.size) { // 1. पीरियड नंबर अपडेट करें document.getElementById('round').innerText = "ID: " + data.next_period; // 2. रिजल्ट (BIG/SMALL) अपडेट करें const resCircle = document.getElementById('result'); resCircle.innerText = data.size; // 3. रंग बदलें (BIG के लिए नारंगी, SMALL के लिए हरा) if (data.size === 'BIG') { resCircle.style.borderColor = '#ff9f00'; resCircle.style.boxShadow = '0 0 20px #ff9f00'; resCircle.style.color = '#ff9f00'; } else { resCircle.style.borderColor = '#00ff00'; resCircle.style.boxShadow = '0 0 20px #00ff00'; resCircle.style.color = '#00ff00'; } } } catch (e) { console.log("Syncing Error:", e); document.getElementById('round').innerText = "ID: RECONNECTING..."; } } // हर 2 सेकंड में डेटा रिफ्रेश करें setInterval(fetchLive, 2000); fetchLive();
